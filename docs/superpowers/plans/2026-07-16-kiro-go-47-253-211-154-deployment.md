```nginx
server {
    listen 80;
    listen [::]:80;
    server_name kiro.dashun9527.com;

    location ^~ /.well-known/acme-challenge/ {
        root /opt/kiro-go/acme-webroot;

        try_files $uri =404;

    error_log /www/wwwlogs/kiro.dashun9527.com.error.log;

# Cloudflare 可能把 HTTP challenge 重定向到 HTTPS。该临时块只在签发前
# 服务 challenge；公网客户端看到的仍是 Cloudflare 边缘证书。
server {
    listen 443 ssl;
    listen [::]:443 ssl;
    http2 on;
    server_name kiro.dashun9527.com;

    ssl_certificate /www/server/panel/vhost/cert/0.default/fullchain.pem;
    ssl_certificate_key /www/server/panel/vhost/cert/0.default/privkey.pem;

    location ^~ /.well-known/acme-challenge/ {
        root /opt/kiro-go/acme-webroot;
        default_type text/plain;
        try_files $uri =404;

        return 404;
    }
}
```

- [ ] **Step 5: 创建独立续期脚本和 timer**

Create `/opt/kiro-go/renew-cert.sh`:

```bash
#!/usr/bin/env bash
set -euo pipefail
cert=/opt/kiro-go/letsencrypt/live/kiro.dashun9527.com/fullchain.pem
before=$(sha256sum "$cert" | awk '{print $1}')
docker run --rm \
  -v /opt/kiro-go/letsencrypt:/etc/letsencrypt \
  -v /opt/kiro-go/acme-webroot:/var/www/acme \
  certbot/certbot@sha256:b4aca0640101294a33b56f44177daaba1d208a250f0e86e84a5033fc1b53bf9f \
  renew --quiet
after=$(sha256sum "$cert" | awk '{print $1}')
if [[ "$before" != "$after" ]]; then
  nginx -t
  nginx -s reload
fi
```

Create `/etc/systemd/system/kiro-go-cert-renew.service`:

```ini
[Unit]
Description=Renew Kiro-Go TLS certificate
After=docker.service network-online.target
Requires=docker.service

[Service]
Type=oneshot
ExecStart=/opt/kiro-go/renew-cert.sh
```

Create `/etc/systemd/system/kiro-go-cert-renew.timer`:

```ini
[Unit]
Description=Daily Kiro-Go TLS certificate renewal check

[Timer]
OnCalendar=daily
RandomizedDelaySec=6h
Persistent=true

[Install]
WantedBy=timers.target
```

Run:

```bash
chmod 0700 /opt/kiro-go/renew-cert.sh
systemctl daemon-reload
systemctl enable --now kiro-go-cert-renew.timer
systemctl is-enabled kiro-go-cert-renew.timer
systemctl is-active kiro-go-cert-renew.timer
```

Expected: timer is enabled and active; no unrelated unit is restarted.
